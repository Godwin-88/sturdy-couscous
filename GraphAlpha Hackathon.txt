  # agent/regime.py
  def detect_regime(prices_df):
      """Returns one of: BullMarket, BearMarket, HighVolatility,
         LowVolatility, SystemicStress, CrisisRegime, RecoveryRegime"""
      # Simple but robust: VIX proxy + 200-day MA + realized vol
      ret = prices_df.pct_change()
      vol_21 = ret.rolling(21).std().iloc[-1] * 252**0.5
      trend  = prices_df.iloc[-1] / prices_df.rolling(200).mean().iloc[-1]

      if vol_21 > 0.30:  return "HighVolatility"
      if trend < 0.95:   return "BearMarket"
      if trend > 1.05:   return "BullMarket"

  # agent/sizing.py
  def kelly_size(signal_strength: float, win_rate: float, 
                 avg_win: float, avg_loss: float) -> float:
      """f* = (p·b - q) / b  where b = avg_win/avg_loss"""
      b = avg_win / avg_loss
      q = 1 - win_rate
      kelly = (win_rate * b - q) / b
      # Use half-Kelly for safety in hackathon
      return min(kelly * 0.5, 0.20)  # cap at 20% per position

  Do You Need Sub-Agents?

  Yes, but fewer than you think. The failure mode of multi-agent systems is premature decomposition — agents that are too fine-grained spend more tokens coordinating than doing useful
  work.

  The right question is: what boundaries justify agent separation?

  An agent boundary is justified when:
  - The task requires a different tool set
  - The task can run in parallel with others
  - The task has a different failure mode that should be isolated
  - The task needs independent memory/context
